-- ============================================================================
-- ai-workflow-automation Core Optimized Database Schema

-- Enable UUID extension
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- ============================================================================
-- CORE USER MANAGEMENT TABLES (3 tables)
-- ============================================================================

-- 1. user_contexts - Used in: app/application/user_service.py, app/application/workflow_service.py
CREATE TABLE IF NOT EXISTS user_contexts (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL,
    context_data JSONB DEFAULT '{}',
    preferences JSONB DEFAULT '{}',
    workflow_patterns JSONB DEFAULT '[]',
    tool_usage_stats JSONB DEFAULT '{}',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    UNIQUE(user_id)
);

-- 2. user_sessions - Used in: app/application/user_service.py
CREATE TABLE IF NOT EXISTS user_sessions (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL,
    session_id VARCHAR(255) NOT NULL UNIQUE,
    session_type VARCHAR(50) NOT NULL,
    connection_id VARCHAR(255),
    context_data JSONB DEFAULT '{}',
    ip_address INET,
    user_agent TEXT,
    is_active BOOLEAN DEFAULT true,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    ended_at TIMESTAMP WITH TIME ZONE,
    expires_at TIMESTAMP WITH TIME ZONE
);

-- 3. conversational_sessions - Used in: app/application/conversational_workflow_service.py
CREATE TABLE IF NOT EXISTS conversational_sessions (
    session_id TEXT PRIMARY KEY,
    user_id TEXT NOT NULL,
    conversation_state JSONB NOT NULL DEFAULT '{}',
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW(),
    archived BOOLEAN DEFAULT false,
    archived_at TIMESTAMPTZ NULL
);

-- ============================================================================
-- WORKFLOW MANAGEMENT TABLES (3 tables)
-- ============================================================================

-- 4. workflows - Used in: app/application/workflow_service.py, app/application/execution_service.py, app/domains/workflow/graphs/graph.py
CREATE TABLE IF NOT EXISTS workflows (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL,
    name VARCHAR(255) NOT NULL,
    description TEXT,
    definition JSONB NOT NULL,
    status VARCHAR(50) DEFAULT 'draft',
    tags TEXT[] DEFAULT '{}',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 5. workflow_executions - Used in: app/application/execution_service.py
CREATE TABLE IF NOT EXISTS workflow_executions (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    workflow_id UUID REFERENCES workflows(id) ON DELETE CASCADE,
    user_id UUID NOT NULL,
    status VARCHAR(50) DEFAULT 'pending',
    test_mode BOOLEAN DEFAULT false,
    input_data JSONB,
    output_data JSONB,
    error_message TEXT,
    started_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    completed_at TIMESTAMP WITH TIME ZONE,
    duration_seconds INTEGER,
    steps_total INTEGER DEFAULT 0,
    steps_completed INTEGER DEFAULT 0,
    steps_failed INTEGER DEFAULT 0
);

-- 6. execution_logs - Used in: app/application/execution_service.py
CREATE TABLE IF NOT EXISTS execution_logs (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    execution_id UUID REFERENCES workflow_executions(id) ON DELETE CASCADE,
    step_id VARCHAR(100),
    step_name VARCHAR(255),
    status VARCHAR(50),
    tool_name VARCHAR(255),
    input_data JSONB,
    output_data JSONB,
    error_message TEXT,
    started_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    completed_at TIMESTAMP WITH TIME ZONE,
    duration_ms INTEGER
);

-- ============================================================================
-- MCP INTEGRATION TABLES (2 tables)
-- ============================================================================

-- MCP Server Definitions Table
CREATE TABLE mcp_servers (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(255) NOT NULL UNIQUE,
    description TEXT,
    access_level VARCHAR(20) NOT NULL CHECK (access_level IN ('system', 'public', 'private')),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- User MCP Server Instances Table
CREATE TABLE user_mcp_servers (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL,
    server_id UUID NOT NULL,
    command JSONB NOT NULL,
    args JSONB,
    env JSONB,
    status VARCHAR(50) NOT NULL DEFAULT 'inactive',
    config JSONB,
    is_active BOOLEAN NOT NULL DEFAULT true,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    
    -- Ensure unique user server instances per user
    UNIQUE(user_id, server_id)
);

-- 7. mcp_tools - Used in: app/application/mcp_service.py, app/application/user_service.py, app/infrastructure/database/providers/supabase.py

-- MCP Tools Table
CREATE TABLE mcp_tools (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(255) NOT NULL,
    description TEXT,
    server_name VARCHAR(100) NOT NULL,
    server_id UUID NOT NULL,
    input_schema JSONB NOT NULL,
    category VARCHAR(100),
    tags JSONB DEFAULT '[]'::jsonb,
    is_active BOOLEAN NOT NULL DEFAULT true,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 8. mcp_server_health - Used in: app/application/mcp_service.py
CREATE TABLE IF NOT EXISTS mcp_server_health (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    server_name VARCHAR(100) NOT NULL,
    is_healthy BOOLEAN NOT NULL,
    response_time_ms INTEGER,
    error_message TEXT,
    check_type VARCHAR(50) DEFAULT 'ping',
    checked_by VARCHAR(100),
    consecutive_failures INTEGER DEFAULT 0,
    last_success_at TIMESTAMP WITH TIME ZONE,
    additional_metrics JSONB DEFAULT '{}',
    last_checked TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    checked_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 9. users

CREATE TABLE IF NOT EXISTS users (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name TEXT,
    email TEXT,
    avatar_url TEXT,
    auth_id TEXT UNIQUE NOT NULL, -- Auth0 user ID (sub claim)
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 9. langgraph_checkpoints - Used in: app/infrastructure/langgraph/checkpointer.py

    checkpoint_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    thread_id VARCHAR(255) NOT NULL,
    workflow_id VARCHAR(255),
    checkpoint_data JSONB NOT NULL,
    metadata JSONB DEFAULT '{}',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),

    -- Loop through all user-defined tables in the public schema
    FOR r IN (SELECT tablename FROM pg_tables WHERE schemaname = 'public') LOOP
        EXECUTE 'DROP TABLE IF EXISTS public.' || quote_ident(r.tablename) || ' CASCADE';
    END LOOP;

    -- Re-enable referential integrity
    EXECUTE 'SET session_replication_role = DEFAULT';
END $$;
